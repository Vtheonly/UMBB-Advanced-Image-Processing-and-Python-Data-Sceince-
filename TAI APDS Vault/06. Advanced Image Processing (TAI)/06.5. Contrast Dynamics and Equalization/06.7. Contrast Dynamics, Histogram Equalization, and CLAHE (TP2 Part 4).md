## 2. Dynamic Range Expansion Function
Remaps pixel intensities linearly to the full $[0, 255]$ scale:

$$Y(y, x) = \text{round}\left( \frac{255}{X_{\max} - X_{\min}} \cdot (X(y, x) - X_{\min}) \right)$$

```python
import cv2
import numpy as np

def expansionHisto(image: np.ndarray, seuil: int = 0, Ymin: int = 0, Ymax: int = 255) -> np.ndarray:
    """Expands dynamic range using noise-tolerant min/max boundaries."""
    hist = np.bincount(image.ravel(), minlength=256)
    valid_indices = np.where(hist > seuil)[0]
    Xmin = valid_indices[0]
    Xmax = valid_indices[-1]
    
    stretched = ((image.astype(np.float64) - Xmin) / (Xmax - Xmin)) * (Ymax - Ymin) + Ymin
    return np.clip(np.round(stretched), Ymin, Ymax).astype(np.uint8)

# Load test images
im1 = cv2.imread("Images/compteur.jpg", cv2.IMREAD_GRAYSCALE)
imc1 = cv2.imread("Images/glaucome-haut-gauche.jpg")

# 1. Contrast Stretching (Custom vs OpenCV)
im2_custom = expansionHisto(im1, seuil=5, Ymin=0, Ymax=255)
im2_opencv = cv2.normalize(im1, None, alpha=0, beta=255, norm_type=cv2.NORM_MINMAX)

# 2. Histogram Equalization: Global vs CLAHE
# Global Equalization
eq_global = cv2.equalizeHist(im1)

# CLAHE (Adaptive Equalization with Contrast Limiting)
clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
eq_clahe = clahe.apply(im1)

# Color Image CLAHE: Process L channel in LAB color space
lab = cv2.cvtColor(imc1, cv2.COLOR_BGR2LAB)
l, a, b = cv2.split(lab)
l_eq = clahe.apply(l)
imc_clahe = cv2.cvtColor(cv2.merge([l_eq, a, b]), cv2.COLOR_LAB2BGR)
